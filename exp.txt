def calc_prob(sentence):
    words = word_tokenize(sentence.lower())
    clean_words = [t for t in words if t not in punctuation]
    prob = 1.0
    print("\nSentence:", sentence)
    print("Bigrams:")
    for pair in ngrams(clean_words, 2):
        bp = bigram_count[pair] / unigram_count[pair[0]] if unigram_count[pair[0]] > 0 else 0

def edit_distance(word1, word2):
    m = len(word1)
    n = len(word2)
    dp = []
    for i in range(m + 1):
        row = []
        for j in range(n + 1):
            row.append(0)
        dp.append(row)
    for i in range(m + 1):
        dp[i][0] = i
    for j in range(n + 1):
        dp[0][j] = j
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if word1[i - 1] == word2[j - 1]:
                dp[i][j] = dp[i - 1][j - 1]
            else:
                dp[i][j] = min(
                    dp[i - 1][j],
                    dp[i][j - 1],
                    dp[i - 1][j - 1]
                ) + 1
    return dp[m][n]

dictionary = [
    "natural", "language", "processing", "learning",
    "text", "correction", "techniques", "machine",
    "computer", "science", "programming", "algorithm"
]

def correct_word(wrong_word):
    best_word = ""
    best_dist = 9999

    for word in dictionary:
        dist = edit_distance(wrong_word, word)
        if dist < best_dist:
            best_dist = dist
            best_word = word

    return best_word, best_dist

wrong_word = "langauge"
corrected, dist = correct_word(wrong_word)

print("Minimum Edit Distance:", dist)

ws = WordNetLemmatizer()
ps = PorterStemmer()

def get_wordnet_pos(tag):
    if tag.startswith("J"):
        return wordnet.ADJ
    elif tag.startswith("V"):
        return wordnet.VERB
    elif tag.startswith("N"):
        return wordnet.NOUN

def generate_words(base_word):
    forms = []
    forms.append(base_word)
    if base_word.endswith("y"):
        forms.append(base_word[:-1] + "ies")
    else:
        forms.append(base_word + "s")
    if base_word.endswith("e"):
        forms.append(base_word + "d")
    else:
        forms.append(base_word + "ed")
    if base_word.endswith("e"):
        forms.append(base_word[:-1] + "ing")
    else:
        forms.append(base_word + "ing")
    return forms

def morphological_analysis(text):
    print("Original Text:")
    print(text)

    tokens = word_tokenize(text.lower())
    print("\n1. Tokenization:")
    print(tokens)

    clean_tokens = [t for t in tokens if t not in stopwords.words("english") and t not in punctuation]
    print("\n2. After Stop Word Removal:")
    print(clean_tokens)

    pos_tags = pos_tag(clean_tokens)
    print("\n3. Morphological Analysis:")
    print("Word\t\tPOS\t\tStem\t\tLemma")
    print("-" * 50)
    lemmas = []
    for word, tag in pos_tags:
        stem = ps.stem(word)
        lemma = ws.lemmatize(word, get_wordnet_pos(tag))
        lemmas.append(lemma)
        print(f"{word:<12}\t{tag:<8}\t{stem:<10}\t{lemma}")

    print("\n4. Word Generation:")
    for lemma in lemmas:
        generated = generate_words(lemma)
        print(lemma, "->", generated)

import nltk
import random

from nltk.corpus import senseval
from nltk.classify import NaiveBayesClassifier
from nltk.classify.util import accuracy
from nltk import word_tokenize, pos_tag

nltk.download('senseval')
nltk.download('punkt')
nltk.download('punkt_tab')
nltk.download('averaged_perceptron_tagger')
nltk.download('averaged_perceptron_tagger_eng')

instances = senseval.instances('hard.pos')
print("Total instances:", len(instances))

def get_features(context, position):
    features = {}
    if position > 0:
        features["previous_word"] = context[position - 1][0].lower()
    else:
        features["previous_word"] = "START"
    if position < len(context) - 1:
        features["next_word"] = context[position + 1][0].lower()
    else:
        features["next_word"] = "END"
    window = context[max(0, position - 2): position] + context[position + 1: position + 3]
    for word in window:
        features["contains_" + word[0].lower()] = True
    return features

featuresets = []
for instance in instances:
    features = get_features(instance.context, instance.position)
    sense = instance.senses[0]
    featuresets.append((features, sense))
random.shuffle(featuresets)

split = int(0.8 * len(featuresets))
train_set = featuresets[:split]
test_set = featuresets[split:]

classifier = NaiveBayesClassifier.train(train_set)

print("\n1. Accuracy:")
print(round(accuracy(classifier, test_set) * 100, 2), "%")

print("\n2. Most Informative Features:")
classifier.show_most_informative_features(10)

def predict_sense(sentence):
    tokens = word_tokenize(sentence)
    tagged_tokens = pos_tag(tokens)
    position = -1
    for i, word in enumerate(tokens):
        if word.lower() == "hard":
            position = i
            break
    if position == -1:
        print("The word 'hard' is not found in the sentence.")
        return
    features = get_features(tagged_tokens, position)
    predicted = classifier.classify(features)
    print("\n3. Predicted Sense:")
    print("Sentence:", sentence)
    print("Predicted Sense:", predicted)

predict_sense("The exam was very hard for the students.")
predict_sense("The stone surface is hard.")
predict_sense("He worked hard to finish the project.")
