def get_pos(tag):

    if tag.startswith('J'):
        return wordnet.ADJ

    elif tag.startswith('V'):
        return wordnet.VERB

    elif tag.startswith('N'):
        return wordnet.NOUN

    elif tag.startswith('R'):
        return wordnet.ADV

    return wordnet.NOUN

def extract_features(word, pos):

    features = []

    # Noun Features
    if pos.startswith('N'):

        features.append("Noun")

        if word.endswith('s'):
            features.append("Plural")

        else:
            features.append("Singular")

    # Verb Features
    elif pos.startswith('V'):

        features.append("Verb")

        if pos == 'VBG':
            features.append("Present Continuous")

        elif pos == 'VBD':
            features.append("Past Tense")

        else:
            features.append("Present Tense")

    # Adjective Features
    elif pos.startswith('J'):

        features.append("Adjective")

    # Adverb Features
    elif pos.startswith('R'):

        features.append("Adverb")

    else:

        features.append("Other")

    return ", ".join(features)

def edit_distance(word1, word2):

    m = len(word1)
    n = len(word2)

    dp = []

    # Create Matrix
    for i in range(m + 1):

        row = []

        for j in range(n + 1):

            row.append(0)

        dp.append(row)

    # Initialize First Column
    for i in range(m + 1):

        dp[i][0] = i

    # Initialize First Row
    for j in range(n + 1):

        dp[0][j] = j

    # Fill Matrix
    for i in range(1, m + 1):

        for j in range(1, n + 1):

            if word1[i - 1] == word2[j - 1]:

                dp[i][j] = dp[i - 1][j - 1]

            else:

                dp[i][j] = min(

                    dp[i - 1][j],      # Deletion
                    dp[i][j - 1],      # Insertion
                    dp[i - 1][j - 1]   # Replacement

                ) + 1

    return dp[m][n]

print("\n===================================")
print("4. Spell Checking using Edit Distance")
print("===================================")

print("Misspelled Word\tSuggested Word\tEdit Distance")
print("-" * 60)

for word in clean_tokens:

    minimum_distance = 999
    corrected_word = word

    for dict_word in dictionary:

        distance = edit_distance(word, dict_word)

        if distance < minimum_distance:

            minimum_distance = distance
            corrected_word = dict_word

    if word != corrected_word:

        print(f"{word:<18}{corrected_word:<18}{minimum_distance}")

import nltk
from string import punctuation

from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk.stem import PorterStemmer
from nltk.util import ngrams

from collections import Counter

nltk.download('punkt')
nltk.download('punkt_tab')
nltk.download('stopwords')

text = """
I love natural language processing.
I love machine learning.
Machine learning is very useful.
Natural language processing is interesting.
I enjoy learning NLP concepts.
I love learning new things.
"""

print("===================================")
print("Original Text")
print("===================================")

print(text)

tokens = word_tokenize(text.lower())

print("\n===================================")
print("1. Tokenization")
print("===================================")

print(tokens)

clean_tokens = [

    t for t in tokens

    if t not in stopwords.words("english")
    and t not in punctuation
]

print("\n===================================")
print("2. After Stop-word Removal")
print("===================================")

print(clean_tokens)

ps = PorterStemmer()

stemmed_tokens = [ps.stem(t) for t in clean_tokens]

print("\n===================================")
print("3. After Stemming")
print("===================================")

print(stemmed_tokens)

unigram_count = Counter(clean_tokens)

bigram_count = Counter(ngrams(clean_tokens, 2))

print("\n===================================")
print("4. Bigram Model")
print("===================================")

for bg, count in bigram_count.items():

    print(bg, "->", count)

print("\n===================================")
print("5. Bigram Probabilities")
print("===================================")

for bg, count in bigram_count.items():

    probability = count / unigram_count[bg[0]]

    print(bg, "=", probability)

def calc_sentence_probability(sentence):

    words = word_tokenize(sentence.lower())

    clean_words = [

        t for t in words

        if t not in stopwords.words("english")
        and t not in punctuation
    ]

    sentence_bigrams = list(ngrams(clean_words, 2))

    probability = 1.0

    print("\nSentence Bigrams and Probabilities:")

    for bg in sentence_bigrams:

        if unigram_count[bg[0]] > 0:

            bg_prob = bigram_count[bg] / unigram_count[bg[0]]

        else:

            bg_prob = 0

        print(bg, "=", bg_prob)

        probability *= bg_prob

    return probability

sentence = input("\nEnter sentence: ")

sentence_probability = calc_sentence_probability(sentence)

print("\n===================================")
print("6. Final Sentence Probability")
print("===================================")

print(sentence_probability)

def predict_next_word(word):

    predictions = []

    total_count = 0

    # Find all bigrams starting with given word
    for (w1, w2), count in bigram_count.items():

        if w1 == word.lower():

            predictions.append((w2, count))

            total_count += count

    return predictions, total_count

print("\n===================================")
print("7. Next Word Prediction")
print("===================================")

input_word = input("Enter a word: ").lower()

results, total = predict_next_word(input_word)

if len(results) > 0:

    print("\nNext Word Probabilities:")

    max_probability = 0
    predicted_word = ""

    for word, count in results:

        probability = count / total

        print(word, "=", probability)

        if probability > max_probability:

            max_probability = probability
            predicted_word = word

    print("\nPredicted Next Word :", predicted_word)

else:

    print("No prediction available.")

# Import Libraries
import nltk

from nltk.corpus import senseval
from nltk.classify import NaiveBayesClassifier
from nltk.classify import accuracy

nltk.download('senseval')

instances = senseval.instances('hard.pos')

print("TOTAL INSTANCES :", len(instances))

sense_meaning = {

    "HARD1": "Difficult",

    "HARD2": "Physically Solid / Intense",

    "HARD3": "Harsh or Severe"
}

def extract_features(instance):

    features = {}

    # Extract context words
    context_words = []

    for item in instance.context:

        word = item[0]

        if isinstance(word, str):
            context_words.append(word.lower())

    # Position of target word "hard"
    target_index = instance.position

    # Window words around target
    window_size = 3

    start = max(0, target_index - window_size)
    end = min(len(context_words), target_index + window_size + 1)

    surrounding_words = context_words[start:end]

    # Create contextual features
    for word in surrounding_words:

        if word.isalpha():
            features[f'contains({word})'] = True

    return features

featuresets = []

for instance in instances:

    features = extract_features(instance)

    sense = instance.senses[0]

    featuresets.append((features, sense))

import random

random.shuffle(featuresets)

train_size = int(len(featuresets) * 0.8)

train_set = featuresets[:train_size]
test_set = featuresets[train_size:]

classifier = NaiveBayesClassifier.train(train_set)

# ------------------------------------------------
# (a) Accuracy
# ------------------------------------------------

acc = accuracy(classifier, test_set)

print("\nACCURACY")
print("-" * 70)

print(f"Accuracy = {acc:.2f}")

# ------------------------------------------------
# (b) Most Informative Features
# ------------------------------------------------

print("\nMOST INFORMATIVE FEATURES")
print("-" * 70)

classifier.show_most_informative_features(10)

# ------------------------------------------------
# (c) Predicted Sense
# ------------------------------------------------

print("\nPREDICTED SENSE")
print("-" * 70)

# Tokenize sentence
words = nltk.word_tokenize(sentence)

# Create Features
test_features = {}

for word in words:

    if word.isalpha():
        test_features[f'contains({word})'] = True

# Predict Sense
predicted_sense = classifier.classify(test_features)

# ------------------------------------------------

print("\nSentence:")
